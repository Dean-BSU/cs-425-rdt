This project focuses on Reliable Data Transfer. We are implementing this over an unreliable protocol. The principles of reliable data transfer we are using are based on Go-Back-N protocol, and includes a checksum, sequence numbers, cumulative acknowledgments, a timer, and a window size.

## Design

`main.c` parses command line arguments for errors and and passes them to `rdt.c`.
`rdt.c` parses command line arguments with getopt for sender/receiver.

The protocol logic is split into three layers, this is done so that protocol logic does not touch a socket, a clock, or a file. 

Layer 1 is packets `packet.c`. This layer just has pure functions, encodes packet into buffer, decodes buffer into packet. 

Layer 2 is Go-Back-N state machine logic `gbn.c`. This contains the sender and receiver machines, each a struct plus the functions that feed it events.

Layer 3 is the socket `io.c`. The only layer that is aware of the socket, relay, clock, or file.

16 Windows is about 15.6x faster than the 1 window scenario. With a larger window size, the sender can send more packets without waiting for an ACK from the receiver, saving the transmission time of 15 50ms ACKs assuming no loss. More processing can occur before the window moves.

The amount of time saved with 0.05 Loss is a lot less, because as window size increases the cost of retransmitting packets upon timeout increases. On each timeout, each packet within the window needs to be resubmitted.

```c

#include "gbn.h"

static rdt_gbn_action_t action(rdt_gbn_action_type_t type, uint32_t sequence,
                               uint8_t retransmission)
{
  /* Returning one value keeps every state transition explicit to the caller. */
  rdt_gbn_action_t result = {type, sequence, retransmission};
  return result;
}

void rdt_gbn_sender_init(rdt_gbn_sender_t *sender, uint32_t total_packets,
                         uint32_t window, uint32_t timeout_ms)
{
  sender->total_packets = total_packets;
  sender->base = 0U;
  sender->next = 0U;
  sender->window = window;
  sender->timeout_ms = timeout_ms;
  sender->timer_started_ms = 0U;
  sender->consecutive_timeouts = 0U;
  sender->timer_active = 0U;
}

rdt_gbn_action_t rdt_gbn_sender_next(rdt_gbn_sender_t *sender,
                                     uint64_t now_ms)
{
  uint8_t retransmission = 0U;
  uint32_t limit;

  if (sender->base == sender->total_packets) {
    return action(RDT_GBN_COMPLETE, 0U, 0U);
  }

  if (sender->timer_active &&
      now_ms - sender->timer_started_ms >= sender->timeout_ms) {
    /* Go-Back-N restarts at base and resends every outstanding packet. */
    sender->consecutive_timeouts++;
    if (sender->consecutive_timeouts >= RDT_GBN_MAX_CONSECUTIVE_TIMEOUTS) {
      return action(RDT_GBN_GIVE_UP, sender->base, 1U);
    }
    sender->next = sender->base;
    sender->timer_started_ms = now_ms;
    retransmission = 1U;
  }

  limit = sender->base + sender->window;
  /* The overflow check protects the window calculation near UINT32_MAX. */
  if (limit < sender->base || limit > sender->total_packets) {
    limit = sender->total_packets;
  }
  if (sender->next < limit) {
    uint32_t sequence = sender->next++;
    if (!sender->timer_active) {
      sender->timer_started_ms = now_ms;
      sender->timer_active = 1U;
    }
    return action(RDT_GBN_SEND, sequence, retransmission);
  }
  return action(RDT_GBN_WAIT, 0U, 0U);
}

void rdt_gbn_sender_ack(rdt_gbn_sender_t *sender, uint32_t sequence,
                        uint64_t now_ms)
{
  if (sequence < sender->base || sequence >= sender->next) {
    /* Old and impossible ACKs cannot move the send window. */
    return;
  }
  sender->base = sequence + 1U;
  sender->consecutive_timeouts = 0U;
  if (sender->base == sender->next) {
    sender->timer_active = 0U;
  } else {
    sender->timer_started_ms = now_ms;
  }
}

void rdt_gbn_receiver_init(rdt_gbn_receiver_t *receiver)
{
  receiver->next_sequence = 0U;
  receiver->fin_sequence = 0U;
  receiver->complete = 0U;
}

rdt_gbn_action_t rdt_gbn_receiver_data(rdt_gbn_receiver_t *receiver,
                                       uint32_t sequence, uint8_t is_fin,
                                       uint64_t now_ms)
{
  (void)now_ms;
  if (receiver->complete) {
    /* A repeated FIN gets the same final ACK, making completion idempotent. */
    return action(RDT_GBN_COMPLETE, receiver->fin_sequence, 0U);
  }
  if (sequence == receiver->next_sequence) {
    if (is_fin != 0U) {
      receiver->fin_sequence = sequence;
      receiver->complete = 1U;
      return action(RDT_GBN_COMPLETE, sequence, 0U);
    }
    /* A gap or duplicate is answered with the last contiguous sequence. */
    receiver->next_sequence++;
    return action(RDT_GBN_SEND, sequence, 0U);
  }
  return action(RDT_GBN_SEND, receiver->next_sequence == 0U
                                     ? 0U
                                     : receiver->next_sequence - 1U,
                1U);
}
```

```c

#include "io.h"
#include "gbn.h"
#include "packet.h"

#include <errno.h>
#include <fcntl.h>
#include <netdb.h>
#include <netinet/in.h>

#include <stdlib.h>
#include <string.h>
#include <sys/socket.h>
#include <time.h>
#include <unistd.h>

#define RDT_MAX_FILE_SIZE (16U * 1024U * 1024U)
#define RDT_REGISTRATION_TIMEOUT_MS 5000U

static uint64_t now_ms(void)
{
  struct timespec time;
  (void)clock_gettime(CLOCK_MONOTONIC, &time);
  /* Monotonic time prevents wall-clock adjustments from breaking timeouts. */
  return (uint64_t)time.tv_sec * 1000U + (uint64_t)time.tv_nsec / 1000000U;
}

static int wait_for_reply(int socket_fd)
{
  struct pollfd descriptor = {socket_fd, POLLIN, 0};
  char reply[128];
  int ready = poll(&descriptor, 1, RDT_REGISTRATION_TIMEOUT_MS);
  ssize_t received;

  if (ready <= 0 || (descriptor.revents & POLLIN) == 0) {
    /* Registration has no peer packet to retransmit, so timeout is failure. */
    return -1;
  }
  received = recv(socket_fd, reply, sizeof(reply) - 1U, 0);
  if (received < 0) {
    return -1;
  }
  reply[received] = '\0';
  if (strncmp(reply, "OK", 2U) != 0) {
    fprintf(stderr, "Relay refused registration: %s\n", reply);
    return -1;
  }
  return 0;
}

static int open_relay_socket(const rdt_config_t *config)
{
  struct addrinfo hints;
  struct addrinfo *addresses = NULL;
  char port[6];
  int socket_fd = -1;
  int result;

  (void)snprintf(port, sizeof(port), "%hu", config->port);
  memset(&hints, 0, sizeof(hints));
  hints.ai_family = AF_UNSPEC;
  hints.ai_socktype = SOCK_DGRAM;
  hints.ai_protocol = IPPROTO_UDP;
  result = getaddrinfo(config->relay, port, &hints, &addresses);
  if (result != 0) {
    fprintf(stderr, "Unable to resolve relay '%s': %s.\n", config->relay,
            gai_strerror(result));
    return -1;
  }
  for (struct addrinfo *address = addresses; address != NULL;
       address = address->ai_next) {
    /* Try each address returned so hostnames work with IPv4 or IPv6. */
    socket_fd = socket(address->ai_family, address->ai_socktype,
                       address->ai_protocol);
    if (socket_fd >= 0 &&
        connect(socket_fd, address->ai_addr, address->ai_addrlen) == 0) {
      break;
    }
    if (socket_fd >= 0) {
      (void)close(socket_fd);
      socket_fd = -1;
    }
  }
  freeaddrinfo(addresses);
  return socket_fd;
}

static int register_with_relay(const rdt_config_t *config, int socket_fd)
{
  char hello[128];
  int length;

  if (config->mode == RDT_MODE_RECEIVE) {
    /* Registration is plain text; packet encoding starts after the OK reply. */
    length = snprintf(hello, sizeof(hello), "HELLO %s recv", config->session);
  } else {
    length = snprintf(hello, sizeof(hello), "HELLO %s send %g %g %g",
                      config->session, config->loss, config->corrupt,
                      config->duplicate);
  }
  if (length < 0 || (size_t)length >= sizeof(hello) ||
      send(socket_fd, hello, (size_t)length, 0) != length) {
    return -1;
  }
  return wait_for_reply(socket_fd);
}

static int sender_run(const rdt_config_t *config, int socket_fd)
{
  FILE *file = fopen(config->file, "rb");
  long file_size;
  uint32_t data_packets;
  uint32_t total_packets;
  rdt_gbn_sender_t sender;
  uint8_t buffer[RDT_PACKET_MAX_SIZE];

  if (file == NULL || fseek(file, 0L, SEEK_END) != 0 ||
      (file_size = ftell(file)) < 0L ||
      (uint64_t)file_size > RDT_MAX_FILE_SIZE ||
      fseek(file, 0L, SEEK_SET) != 0) {
    fprintf(stderr, "Unable to read input file '%s'.\n", config->file);
    if (file != NULL) {
      (void)fclose(file);
    }
    return -1;
  }
  data_packets = (uint32_t)(((uint64_t)file_size + RDT_PACKET_MAX_PAYLOAD - 1U) /
                            RDT_PACKET_MAX_PAYLOAD);
  /* FIN occupies one sequence number, including for an empty file. */
  total_packets = data_packets + 1U;
  rdt_gbn_sender_init(&sender, total_packets, config->window,
                      config->timeout_ms);

  for (;;) {
    rdt_gbn_action_t action = rdt_gbn_sender_next(&sender, now_ms());
    if (action.type == RDT_GBN_COMPLETE) {
      (void)fclose(file);
      return 0;
    }
    if (action.type == RDT_GBN_GIVE_UP) {
      fprintf(stderr, "Transfer gave up after %u consecutive timeouts.\n",
              RDT_GBN_MAX_CONSECUTIVE_TIMEOUTS);
      (void)fclose(file);
      return -1;
    }
    if (action.type == RDT_GBN_SEND) {
      rdt_packet_t packet = {0};
      size_t wire_length;
      packet.type = action.sequence == data_packets ? RDT_PACKET_FIN
                                                     : RDT_PACKET_DATA;
      packet.sequence = action.sequence;
      if (packet.type == RDT_PACKET_DATA) {
        /* Seek from the sequence number so retransmissions need no cache. */
        if (fseek(file, (long)action.sequence * RDT_PACKET_MAX_PAYLOAD,
                  SEEK_SET) != 0) {
          (void)fclose(file);
          return -1;
        }
        packet.length = (uint16_t)fread(packet.payload, 1U,
                                        RDT_PACKET_MAX_PAYLOAD, file);
        if (ferror(file) != 0) {
          (void)fclose(file);
          return -1;
        }
      }
      if (rdt_packet_encode(&packet, buffer, sizeof(buffer), &wire_length) != 0 ||
          send(socket_fd, buffer, wire_length, 0) != (ssize_t)wire_length) {
        (void)fclose(file);
        return -1;
      }
    } else {
      /* Waiting is the state-machine's request to poll for an ACK. */
      struct pollfd descriptor = {socket_fd, POLLIN, 0};
      if (poll(&descriptor, 1, (int)config->timeout_ms) < 0 &&
          errno != EINTR) {
        (void)fclose(file);
        return -1;
      }
      if ((descriptor.revents & POLLIN) != 0) {
        ssize_t received = recv(socket_fd, buffer, sizeof(buffer), 0);
        rdt_packet_t ack;
        if (received > 0 &&
            rdt_packet_decode(buffer, (size_t)received, &ack) == 0 &&
            ack.type == RDT_PACKET_ACK) {
          rdt_gbn_sender_ack(&sender, ack.sequence, now_ms());
        }
      }
    }
  }
}

static int receiver_run(const rdt_config_t *config, int socket_fd)
{
  FILE *file = fopen(config->file, "wb");
  rdt_gbn_receiver_t receiver;
  uint8_t buffer[RDT_PACKET_MAX_SIZE];

  if (file == NULL) {
    fprintf(stderr, "Unable to open output file '%s'.\n", config->file);
    return -1;
  }
  rdt_gbn_receiver_init(&receiver);
  for (;;) {
    ssize_t received = recv(socket_fd, buffer, sizeof(buffer), 0);
    rdt_packet_t packet;
    if (received <= 0 || rdt_packet_decode(buffer, (size_t)received, &packet) !=
                              0 ||
        packet.type == RDT_PACKET_ACK) {
      /* Invalid datagrams and ACKs are not receiver data; ignore them. */
      continue;
    }
    uint8_t in_order = packet.sequence == receiver.next_sequence;
    rdt_gbn_action_t action = rdt_gbn_receiver_data(
        &receiver, packet.sequence, packet.type == RDT_PACKET_FIN, now_ms());
    if (in_order && packet.type == RDT_PACKET_DATA &&
        fwrite(packet.payload, 1U, packet.length, file) != packet.length) {
      /* Only the next contiguous packet may be delivered to the file. */
      (void)fclose(file);
      return -1;
    }
    rdt_packet_t ack = {.type = RDT_PACKET_ACK, .sequence = action.sequence};
    size_t wire_length;
    if (rdt_packet_encode(&ack, buffer, sizeof(buffer), &wire_length) != 0 ||
        send(socket_fd, buffer, wire_length, 0) != (ssize_t)wire_length) {
      (void)fclose(file);
      return -1;
    }
    if (action.type == RDT_GBN_COMPLETE) {
      (void)fclose(file);
      return 0;
    }
  }
}

int rdt_run(const rdt_config_t *config)
{
  int socket_fd;
  int result;

  if (config == NULL) {
    fprintf(stderr, "Transfer configuration is missing.\n");
    return -1;
  }
  socket_fd = open_relay_socket(config);
  if (socket_fd < 0 || register_with_relay(config, socket_fd) != 0) {
    if (socket_fd >= 0) {
      (void)close(socket_fd);
    }
    return -1;
  }
  result = config->mode == RDT_MODE_SEND ? sender_run(config, socket_fd)
                                         : receiver_run(config, socket_fd);
  (void)close(socket_fd);
  return result;
}
```

```c
#include "packet.h"

#include <arpa/inet.h>
#include <string.h>

static int valid_type(uint8_t type)
{
  /* The enum values are deliberately contiguous from DATA through FIN. */
  return type <= (uint8_t)RDT_PACKET_FIN;
}

uint16_t rdt_internet_checksum(const uint8_t *data, size_t length)
{
  uint32_t sum = 0;

  while (length >= 2U) {
    /* The Internet checksum adds 16-bit words in network-byte order. */
    sum += ((uint32_t)data[0] << 8U) | (uint32_t)data[1];
    data += 2;
    length -= 2U;
  }
  if (length != 0U) {
    /* RFC 1071 pads an odd trailing byte on the right. */
    sum += (uint32_t)data[0] << 8U;
  }

  while ((sum >> 16U) != 0U) {
    /* Fold carries back into the low 16 bits until none remain. */
    sum = (sum & 0xffffU) + (sum >> 16U);
  }
  return (uint16_t)~sum;
}

int rdt_packet_encode(const rdt_packet_t *packet, uint8_t *wire,
                      size_t wire_capacity, size_t *wire_length)
{
  uint16_t network_length;
  uint32_t network_sequence;
  size_t total_length;
  uint16_t checksum;

  if (packet == NULL || wire == NULL || wire_length == NULL ||
      !valid_type((uint8_t)packet->type) ||
      packet->length > RDT_PACKET_MAX_PAYLOAD ||
      ((packet->type == RDT_PACKET_ACK || packet->type == RDT_PACKET_FIN) &&
       packet->length != 0U)) {
    return -1;
  }

  /* Reserve checksum bytes while calculating the checksum over the packet. */
  total_length = RDT_PACKET_HEADER_SIZE + (size_t)packet->length;
  if (wire_capacity < total_length) {
    return -1;
  }

  wire[0] = (uint8_t)packet->type;
  wire[1] = 0U;
  wire[2] = 0U;
  wire[3] = 0U;
  network_sequence = htonl(packet->sequence);
  memcpy(wire + 4, &network_sequence, sizeof(network_sequence));
  network_length = htons(packet->length);
  memcpy(wire + 8, &network_length, sizeof(network_length));
  if (packet->length != 0U) {
    memcpy(wire + RDT_PACKET_HEADER_SIZE, packet->payload, packet->length);
  }

  checksum = rdt_internet_checksum(wire, total_length);
  wire[2] = (uint8_t)(checksum >> 8U);
  wire[3] = (uint8_t)(checksum & 0xffU);
  *wire_length = total_length;
  return 0;
}

int rdt_packet_decode(const uint8_t *wire, size_t wire_length,
                      rdt_packet_t *packet)
{
  uint16_t network_length;
  uint32_t network_sequence;
  size_t expected_length;

  if (wire == NULL || packet == NULL || wire_length < RDT_PACKET_HEADER_SIZE ||
      wire_length > RDT_PACKET_MAX_SIZE || !valid_type(wire[0]) ||
      wire[1] != 0U || rdt_internet_checksum(wire, wire_length) != 0U) {
    return -1;
  }

  /* The declared payload length determines the only valid datagram size. */
  memcpy(&network_length, wire + 8, sizeof(network_length));
  packet->length = ntohs(network_length);
  expected_length = RDT_PACKET_HEADER_SIZE + (size_t)packet->length;
  if (packet->length > RDT_PACKET_MAX_PAYLOAD ||
      expected_length != wire_length ||
      ((wire[0] == (uint8_t)RDT_PACKET_ACK ||
        wire[0] == (uint8_t)RDT_PACKET_FIN) &&
       packet->length != 0U)) {
    return -1;
  }

  memcpy(&network_sequence, wire + 4, sizeof(network_sequence));
  packet->type = (rdt_packet_type_t)wire[0];
  packet->sequence = ntohl(network_sequence);
  if (packet->length != 0U) {
    memcpy(packet->payload, wire + RDT_PACKET_HEADER_SIZE, packet->length);
  }
  return 0;
}
```

```c
static int parse_unsigned(const char *text, unsigned long maximum,
                          unsigned long *value)
{
  char *end = NULL;
  unsigned long parsed;

  // GCOVR_EXCL_START
  if (text == NULL || *text == '\0' || text[0] == '-') {
    return -1;
  }
  // GCOVR_EXCL_STOP

  errno = 0;
  parsed = strtoul(text, &end, 10);
  if (errno == ERANGE || end == text || *end != '\0' || parsed > maximum) {
    return -1;
  }

  *value = parsed;
  return 0;
}

static int parse_probability(const char *text, double *value)
{
  char *end = NULL;
  double parsed;

  if (text == NULL || *text == '\0') {
    return -1;
  }

  errno = 0;
  parsed = strtod(text, &end);
  if (errno == ERANGE || end == text || *end != '\0' || parsed < 0.0 ||
      parsed > 1.0) {
    return -1;
  }

  *value = parsed;
  return 0;
}

int rdt_parse_args(int argc, char *argv[], rdt_config_t *config)
{
  int option;
  unsigned long parsed;
  const char *value;

  if (config == NULL || argc < 2) {
    return -1;
  }

  memset(config, 0, sizeof(*config));
  config->window = RDT_DEFAULT_WINDOW;
  config->timeout_ms = RDT_DEFAULT_TIMEOUT_MS;
  config->port = RDT_DEFAULT_RELAY_PORT;

  if (strcmp(argv[1], "send") == 0) {
    config->mode = RDT_MODE_SEND;
  } else if (strcmp(argv[1], "recv") == 0) {
    config->mode = RDT_MODE_RECEIVE;
  } else {
    fprintf(stderr, "Expected mode 'send' or 'recv', got '%s'.\n", argv[1]);
    return -1;
  }

  optind = 1;
  while ((option = getopt(argc - 1, argv + 1, ":s:w:T:l:c:d:p:")) != -1) {
    switch (option) {
    case 's':
      config->session = optarg;
      break;
    case 'w':
      if (config->mode == RDT_MODE_RECEIVE) {
        fprintf(stderr, "Option -w is only available in sender mode.\n");
        return -1;
      }
      if (parse_unsigned(optarg, 64, &parsed) != 0 || parsed == 0) {
        fprintf(stderr, "Window must be an integer from 1 through 64.\n");
        return -1;
      }
      config->window = (unsigned int)parsed;
      break;
    case 'T':
      if (config->mode == RDT_MODE_RECEIVE) {
        fprintf(stderr, "Option -T is only available in sender mode.\n");
        return -1;
      }
      if (parse_unsigned(optarg, UINT_MAX, &parsed) != 0 || parsed == 0) {
        fprintf(stderr, "Timeout must be a positive integer in milliseconds.\n");
        return -1;
      }
      config->timeout_ms = (unsigned int)parsed;
      break;
    case 'l':
      if (parse_probability(optarg, &config->loss) != 0) {
        fprintf(stderr, "Loss must be a probability from 0 through 1.\n");
        return -1;
      }
      break;
    case 'c':
      if (parse_probability(optarg, &config->corrupt) != 0) {
        fprintf(stderr, "Corruption must be a probability from 0 through 1.\n");
        return -1;
      }
      break;
    case 'd':
      if (parse_probability(optarg, &config->duplicate) != 0) {
        fprintf(stderr, "Duplication must be a probability from 0 through 1.\n");
        return -1;
      }
      break;
    case 'p':
      if (parse_unsigned(optarg, 65535, &parsed) != 0 || parsed == 0) {
        fprintf(stderr, "Port must be an integer from 1 through 65535.\n");
        return -1;
      }
      config->port = (unsigned short)parsed;
      break;
    case ':':
      fprintf(stderr, "Option %s requires a value.\n", argv[optind]);
      return -1;
    case '?':
      fprintf(stderr, "Unknown option: -%c.\n", optopt);
      return -1;
    // GCOVR_EXCL_START
    default:
      return -1;
    // GCOVR_EXCL_STOP
    }
  }

  if (config->session == NULL || config->session[0] == '\0') {
    fprintf(stderr, "A non-empty session is required with -s.\n");
    return -1;
  }
  if (argc - 1 - optind != 2) {
    fprintf(stderr, "Expected <relay> and <file>.\n");
    return -1;
  }

  value = argv[optind + 1];
  if (value[0] == '\0') {
    fprintf(stderr, "Relay address cannot be empty.\n");
    return -1;
  }
  config->relay = value;
  config->file = argv[optind + 2];
  if (config->file[0] == '\0') {
    fprintf(stderr, "File path cannot be empty.\n");
    return -1;
  }

  if (config->mode == RDT_MODE_RECEIVE &&
      (config->loss != 0.0 || config->corrupt != 0.0 ||
       config->duplicate != 0.0)) {
    fprintf(stderr, "Loss, corruption, and duplication are sender options.\n");
    return -1;
  }

  return 0;
}
```

```c
#define SIM_QUEUE_SIZE 4096U
#define SIM_MAX_STEPS 1000000U
#define SIM_FAILURE (-1)

typedef struct {
  uint8_t bytes[RDT_PACKET_MAX_SIZE];
  size_t length;
} datagram_t;

typedef struct {
  datagram_t packets[SIM_QUEUE_SIZE];
  size_t count;
} queue_t;

static uint32_t random_next(uint32_t *state)
{
  uint32_t value = *state;
  value ^= value << 13U;
  value ^= value >> 17U;
  value ^= value << 5U;
  *state = value == 0U ? 0x6d2b79f5U : value;
  return *state;
}

static int enqueue(queue_t *queue, const uint8_t *bytes, size_t length,
                   uint32_t *random)
{
  datagram_t *packet;

  if (queue->count == SIM_QUEUE_SIZE || length > RDT_PACKET_MAX_SIZE) {
    return SIM_FAILURE;
  }
  packet = &queue->packets[queue->count++];
  memcpy(packet->bytes, bytes, length);
  packet->length = length;

  if (random_next(random) % 100U < 20U) {
    queue->count--;
    return 0;
  }
  if (random_next(random) % 100U < 20U) {
    packet->bytes[random_next(random) % length] ^=
        (uint8_t)(1U << (random_next(random) % 8U));
  } else if (random_next(random) % 100U < 20U) {
    if (queue->count == SIM_QUEUE_SIZE) {
      return SIM_FAILURE;
    }
    queue->packets[queue->count++] = *packet;
  }
  return 0;
}

static int dequeue(queue_t *queue, datagram_t *packet)
{
  if (queue->count == 0U) {
    return 0;
  }
  *packet = queue->packets[0];
  memmove(queue->packets, queue->packets + 1U,
          (queue->count - 1U) * sizeof(queue->packets[0]));
  queue->count--;
  return 1;
}

int rdt_simulate_transfer(uint32_t seed)
{
  static queue_t to_receiver;
  static queue_t to_sender;
  uint8_t source[SIM_PAYLOAD_SIZE];
  uint8_t received[SIM_PAYLOAD_SIZE];
  rdt_gbn_sender_t sender;
  rdt_gbn_receiver_t receiver;
  uint32_t random = seed == 0U ? 1U : seed;
  uint32_t total_packets =
      (SIM_PAYLOAD_SIZE + RDT_PACKET_MAX_PAYLOAD - 1U) /
          RDT_PACKET_MAX_PAYLOAD +
      1U;
  size_t received_length = 0U;
  uint64_t now = 0U;

  memset(&to_receiver, 0, sizeof(to_receiver));
  memset(&to_sender, 0, sizeof(to_sender));
  for (size_t index = 0U; index < sizeof(source); index++) {
    source[index] = (uint8_t)(index * 29U + 7U);
  }
  rdt_gbn_sender_init(&sender, total_packets, 8U, 25U);
  rdt_gbn_receiver_init(&receiver);

  for (uint32_t step = 0U; step < SIM_MAX_STEPS; step++) {
    rdt_gbn_action_t sender_action = rdt_gbn_sender_next(&sender, now);
    if (sender_action.type == RDT_GBN_GIVE_UP) {
      return SIM_FAILURE;
    }
    if (sender_action.type == RDT_GBN_COMPLETE) {
      return receiver.complete != 0U && received_length == sizeof(source) &&
                     memcmp(source, received, sizeof(source)) == 0
                 ? 0
                 : SIM_FAILURE;
    }
    if (sender_action.type == RDT_GBN_SEND) {
      rdt_packet_t packet = {0};
      uint8_t wire[RDT_PACKET_MAX_SIZE];
      size_t wire_length;
      uint32_t data_packets = total_packets - 1U;

      packet.type = sender_action.sequence == data_packets
                        ? RDT_PACKET_FIN
                        : RDT_PACKET_DATA;
      packet.sequence = sender_action.sequence;
      if (packet.type == RDT_PACKET_DATA) {
        size_t offset = (size_t)packet.sequence * RDT_PACKET_MAX_PAYLOAD;
        packet.length = (uint16_t)((sizeof(source) - offset) >
                                           RDT_PACKET_MAX_PAYLOAD
                                       ? RDT_PACKET_MAX_PAYLOAD
                                       : sizeof(source) - offset);
        memcpy(packet.payload, source + offset, packet.length);
      }
      if (rdt_packet_encode(&packet, wire, sizeof(wire), &wire_length) != 0 ||
          enqueue(&to_receiver, wire, wire_length, &random) != 0) {
        return SIM_FAILURE;
      }
    }

    datagram_t datagram;
    if (dequeue(&to_receiver, &datagram) != 0) {
      rdt_packet_t packet;
      if (rdt_packet_decode(datagram.bytes, datagram.length, &packet) == 0 &&
          packet.type != RDT_PACKET_ACK) {
        uint8_t in_order = packet.sequence == receiver.next_sequence;
        rdt_gbn_action_t receiver_action = rdt_gbn_receiver_data(
            &receiver, packet.sequence, packet.type == RDT_PACKET_FIN, now);
        if (in_order && packet.type == RDT_PACKET_DATA) {
          if (received_length + packet.length > sizeof(received)) {
            return SIM_FAILURE;
          }
          memcpy(received + received_length, packet.payload, packet.length);
          received_length += packet.length;
        }
        if (receiver_action.type == RDT_GBN_COMPLETE ||
            receiver_action.type == RDT_GBN_SEND) {
          rdt_packet_t ack = {
              .type = RDT_PACKET_ACK, .sequence = receiver_action.sequence};
          uint8_t wire[RDT_PACKET_MAX_SIZE];
          size_t wire_length;
          if (rdt_packet_encode(&ack, wire, sizeof(wire), &wire_length) != 0 ||
              enqueue(&to_sender, wire, wire_length, &random) != 0) {
            return SIM_FAILURE;
          }
        }
      }
    }

    if (dequeue(&to_sender, &datagram) != 0) {
      rdt_packet_t ack;
      if (rdt_packet_decode(datagram.bytes, datagram.length, &ack) == 0 &&
          ack.type == RDT_PACKET_ACK) {
        rdt_gbn_sender_ack(&sender, ack.sequence, now);
      }
    }
    if (sender_action.type == RDT_GBN_WAIT || to_receiver.count == 0U) {
      now += 1U;
    }
  }
  return SIM_FAILURE;
}
```
